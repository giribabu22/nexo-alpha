```text
@nexo/cli
     │

     │            │

     │            │

     │            │
     └─────── @nexo/core
                    │
                  Node.js
```

The important rule is:

> **`@nexo/core` must not depend on AI, Hapi, CLI, database libraries, or cloud services.**

# 9. Application

Create:

```text
packages/core/src/application.ts
```

```ts
import type { NexoModule } from "./module.js";

export interface ApplicationOptions {
  readonly name: string;
  readonly version?: string;
  readonly description?: string;
}

export type ApplicationState =
  | "created"
  | "initializing"
  | "running"
  | "stopping"
  | "stopped";

export class NexoApplication {
  readonly name: string;
  readonly version: string;
  readonly description?: string;

  private readonly modules = new Map<string, NexoModule>();

  private _state: ApplicationState = "created";

  constructor(options: ApplicationOptions) {
    this.name = options.name;
    this.version = options.version ?? "0.1.0";
    this.description = options.description;
  }

  get state(): ApplicationState {
    return this._state;
  }

  module(module: NexoModule): this {
    if (this.modules.has(module.name)) {
      throw new Error(
        `Nexo module "${module.name}" is already registered.`
      );
    }

    this.modules.set(module.name, module);

    return this;
  }

  getModule(name: string): NexoModule | undefined {
    return this.modules.get(name);
  }

  getModules(): readonly NexoModule[] {
    return [...this.modules.values()];
  }

  async start(): Promise<void> {
    if (this._state === "running") {
      return;
    }

    if (this._state !== "created" && this._state !== "stopped") {
      throw new Error(
        `Cannot start application while in "${this._state}" state.`
      );
    }

    this._state = "initializing";

    for (const module of this.modules.values()) {
      await module.initialize?.();
    }

    for (const module of this.modules.values()) {
      await module.start?.();
    }

    this._state = "running";
  }

  async stop(): Promise<void> {
    if (this._state === "stopped") {
      return;
    }

    if (this._state !== "running") {
      throw new Error(
        `Cannot stop application while in "${this._state}" state.`
      );
    }

    this._state = "stopping";

    const modules = [...this.modules.values()].reverse();

    for (const module of modules) {
      await module.stop?.();
    }

    this._state = "stopped";
  }
}

export function createApplication(
  options: ApplicationOptions
): NexoApplication {
  return new NexoApplication(options);
}
```

This gives us the first real Nexo runtime.

# 12. Nexo errors

Create:

```text
packages/core/src/errors.ts
```

```ts
export class NexoError extends Error {
  readonly code: string;

  constructor(code: string, message: string) {
    super(message);

    this.name = "NexoError";
    this.code = code;
  }
}

export class NexoConfigurationError extends NexoError {
  constructor(message: string) {
    super("NEXO_CONFIGURATION_ERROR", message);
    this.name = "NexoConfigurationError";
  }
}

export class NexoLifecycleError extends NexoError {
  constructor(message: string) {
    super("NEXO_LIFECYCLE_ERROR", message);
    this.name = "NexoLifecycleError";
  }
}
```

We'll eventually want a proper error hierarchy, but this is enough to establish the concept.

# 14. First test

Create:

```text
packages/core/test/application.test.js
```

For the very first test, we can test the compiled package with Node's built-in test runner.

```js
import test from "node:test";
import assert from "node:assert/strict";

import { createApplication } from "../dist/index.js";

test("creates a Nexo application", () => {
  const app = createApplication({
    name: "shop"
  });

  assert.equal(app.name, "shop");
  assert.equal(app.version, "0.1.0");
  assert.equal(app.state, "created");
});

test("registers modules", () => {
  const app = createApplication({
    name: "shop"
  });

  app.module({
    name: "payments",
    description: "Handle customer payments"
  });

  assert.equal(app.getModules().length, 1);
  assert.equal(app.getModule("payments")?.name, "payments");
});

test("runs module lifecycle", async () => {
  const events = [];

  const app = createApplication({
    name: "shop"
  });

  app.module({
    name: "payments",

    initialize() {
      events.push("initialize");
    },

    start() {
      events.push("start");
    },

    stop() {
      events.push("stop");
    }
  });

  await app.start();

  assert.equal(app.state, "running");
  assert.deepEqual(events, [
    "initialize",
    "start"
  ]);

  await app.stop();

  assert.equal(app.state, "stopped");
  assert.deepEqual(events, [
    "initialize",
    "start",
    "stop"
  ]);
});
```

## 18. The next architectural step

**Do not jump directly to Hapi.**

The next thing I'd build is the **Nexo internal application model**.

Right now:

```text
Application
   │
   └── Modules
```

We want:

```text
Application
│
├── Modules
│   ├── Users
│   ├── Orders
│   └── Payments
│
├── APIs
├── Services
├── Dependencies
├── Configuration
└── Context
```

And eventually:

```text
Application
│
├── Identity
├── Architecture
│   ├── Modules
│   ├── APIs
│   ├── Services
│   └── Dependencies
│
├── Business Context
│   ├── Requirements
│   └── Rules
│
├── Development State
│   ├── Current Work
│   ├── Completed
│   ├── In Progress
│   └── Blocked
│
├── Decisions
├── Constraints
└── History
```

**That is the part that makes Nexo different.**

Hapi is infrastructure. The Nexo application model is the product.

So I recommend this exact build sequence:

```text
STEP 1  Repository
   ↓
STEP 2  @nexo/core
   ↓
STEP 3  Application lifecycle       ← we are here
   ↓
STEP 4  Module model
   ↓
STEP 5  Service model
   ↓
STEP 6  API model
   ↓
STEP 7  Dependency model
   ↓
STEP 8  Configuration
   ↓
STEP 9  Application Context          ← Nexo's key differentiator
   ↓
STEP 10 Hapi adapter
   ↓
STEP 11 CLI
   ↓
STEP 12 Inspect / Status
   ↓
STEP 13 AI read interface
   ↓
STEP 14 Safe AI operations
```

**The most important design decision right now:** keep `@nexo/core` framework-level and provider-independent. If we get the core application model right, Hapi, CLI, MCP, AI agents, IDEs, and other tooling can all sit on top of it instead of becoming the framework itself.
